class NotificationView extends StatelessWidget {
  final _filter = NotifFilter.all.obs;

  List<NotificationModel> _filtered(List<NotificationModel> all) {
    switch (_filter.value) {
      case NotifFilter.unread: return all.where((n) => !n.isRead).toList();

  Widget _buildCouponCard(BuildContext context, CouponModel coupon) {
    final discountText = coupon.code == 'FREESHIP' ? 'FREE'
        : coupon.isPercentage
          ? '${coupon.discountAmount.toInt()}%'
          : '\$${coupon.discountAmount.toInt()}';
    return Container(
      decoration: BoxDecoration(color: Colors.white,
        borderRadius: BorderRadius.circular(16),
        border: Border.all(color: Color(0xFFECEFF2))),
      child: Row(children: [
        // Left discount badge section (big % or $ amount)
        Container(width: 90, child: Text(discountText)),
        // Right: coupon code (copyable), description, expiry, Apply button
        Expanded(child: Column(children: [
          Text(coupon.code, style: AppTextStyles.couponCodeStyle),
          Text(coupon.description),
          Text('Min: \$${coupon.minPurchase}'),

      ]),
    );
