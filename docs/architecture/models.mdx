# 📊 Data Models

All data models live in `lib/models/`. They use `fromJson` / `toJson` for serialization and include `copyWith` for immutable updates.

---

## 📁 Files

```
lib/models/
├── product_model.dart
├── cart_item_model.dart
├── user_model.dart
├── coupon_model.dart
├── notification_model.dart
└── hana_ai_message.dart
```

:::info Seed-Based Data
Some entities (addresses, orders, payment cards, delivery slots) are stored as `Map<String, dynamic>` in seed data files rather than dedicated model classes. This keeps the template lightweight — add formal models when connecting to a real backend.
:::

## 🛒 ProductModel

The core model used across all catalog, cart, and AI screens:

```dart
// lib/models/product_model.dart

enum ProductCategory {
  vegetables, snacks, fruits, diary,
  pantry, beverages, frozen, household,
}

class ProductModel {
  final String id;
  final String name;
  final double price;
  final double originalPrice;
  final bool inStock;
  final String assetPath;
  final IconData? icon;
  final Color? color;
  final List<String> tags;
  final String description;
  final String sizeText;
  final String? categoryLabel;

  ProductModel({
    required this.id,
    required this.name,
    required this.price,
    required this.originalPrice,
    required this.inStock,
    required this.assetPath,
    this.icon,
    this.color,
    required this.tags,
    required this.description,
    required this.sizeText,
    this.categoryLabel,
  });

  factory ProductModel.fromJson(Map<String, dynamic> json) {
    return ProductModel(
      id: json['id'] as String? ?? '',
      name: json['name'] as String? ?? '',
      price: (json['price'] as num?)?.toDouble() ?? 0.0,
      originalPrice: (json['originalPrice'] as num?)?.toDouble() ?? 0.0,
      inStock: json['inStock'] as bool? ?? true,
      assetPath: json['assetPath'] as String? ?? '',
      icon: json['icon'] as IconData?,
      color: json['color'] as Color?,
      tags: (json['tags'] as List<dynamic>?)
          ?.map((e) => e as String).toList() ?? [],
      description: json['description'] as String? ?? '',
      sizeText: json['sizeText'] as String? ?? '',
      categoryLabel: json['categoryLabel'] as String?,
    );
  }

  Map<String, dynamic> toJson() => {
    'id': id, 'name': name, 'price': price,
    'originalPrice': originalPrice, 'inStock': inStock,
    'assetPath': assetPath, 'tags': tags,
    'description': description, 'sizeText': sizeText,
    'categoryLabel': categoryLabel,
  };

  // ── Computed Properties ──────────────────────────────
  double get discountPercentage {
    if (originalPrice <= 0) return 0.0;
    return ((originalPrice - price) / originalPrice) * 100;
  }

  String get formattedPrice => '₹${price.toStringAsFixed(2)}';
  String get formattedOriginalPrice => '₹${originalPrice.toStringAsFixed(2)}';

  ProductCategory? get category {
    if (categoryLabel == null) return null;
    try {
      return ProductCategory.values.firstWhere(
        (e) => e.name == categoryLabel!.toLowerCase(),
      );
    } catch (e) { return null; }
  }
}
```

// lib/models/coupon_model.dart

class CouponModel {
  final String code;
  final double discountAmount;
  final bool isPercentage;
  final double minPurchase;
  final String description;
  final String expiryDate;

  const CouponModel({
    required this.code,
    required this.discountAmount,
    required this.isPercentage,
    required this.minPurchase,
    required this.description,
    required this.expiryDate,
  });

  /// Calculates the discount for a given subtotal and shipping charge
  double calculateDiscount(double subtotal, double shippingCharge) {
    if (subtotal < minPurchase) return 0.0;
    if (code == 'FREESHIP') return shippingCharge;
    if (isPercentage) return subtotal * (discountAmount / 100.0);
    return discountAmount;
  }
}

// lib/models/product_model.dart

class ProductModel {
  final String id;
  final String name;
  final String brand;
  final String category;
  final String subcategory;
  final String imageUrl;
  final List<String> imageUrls;      // Gallery images
  final double price;
  final double? originalPrice;       // null = no discount
  final String weight;               // e.g. "500g", "1kg"
  final List<String> variants;       // e.g. ["250g", "500g", "1kg"]
  final double rating;               // 0.0 – 5.0
  final int reviewCount;
  final bool inStock;
  final int stockQuantity;
  final String description;
  final List<String> tags;
  final bool isFeatured;
  final bool isAiRecommended;

  const ProductModel({
    required this.id,
    required this.name,
    required this.brand,
    required this.category,
    this.subcategory = '',
    required this.imageUrl,
    this.imageUrls = const [],
    required this.price,
    this.originalPrice,
    this.weight = '',
    this.variants = const [],
    this.rating = 0.0,
    this.reviewCount = 0,
    this.inStock = true,
    this.stockQuantity = 0,
    this.description = '',
    this.tags = const [],
    this.isFeatured = false,
    this.isAiRecommended = false,
  });

  double get discountPercent {
    if (originalPrice == null || originalPrice! <= price) return 0;
    return ((originalPrice! - price) / originalPrice! * 100).roundToDouble();
  }

  bool get hasDiscount => originalPrice != null && originalPrice! > price;

  factory ProductModel.fromJson(Map<String, dynamic> json) => ProductModel(
    id: json['id'] as String,
    name: json['name'] as String,
    brand: json['brand'] as String? ?? '',
    category: json['category'] as String,
    imageUrl: json['imageUrl'] as String,
    imageUrls: (json['imageUrls'] as List?)?.cast<String>() ?? [],
    price: (json['price'] as num).toDouble(),
    originalPrice: json['originalPrice'] != null
        ? (json['originalPrice'] as num).toDouble()
        : null,
    weight: json['weight'] as String? ?? '',
    variants: (json['variants'] as List?)?.cast<String>() ?? [],
    rating: (json['rating'] as num?)?.toDouble() ?? 0.0,
    reviewCount: json['reviewCount'] as int? ?? 0,
    inStock: json['inStock'] as bool? ?? true,
    stockQuantity: json['stockQuantity'] as int? ?? 0,
    description: json['description'] as String? ?? '',
  );

  Map<String, dynamic> toJson() => {
    'id': id, 'name': name, 'brand': brand,
    'category': category, 'imageUrl': imageUrl,
    'price': price, 'originalPrice': originalPrice,
    'weight': weight, 'inStock': inStock,
  };
}

## 🎟️ CouponModel

```dart
// lib/models/coupon_model.dart

enum DiscountType { percentage, fixed }

class CouponModel {
  final String id;
  final String code;
  final String title;
  final String description;
  final DiscountType discountType;
  final double discountValue;     // % or fixed amount
  final double minOrderValue;
  final double? maxDiscount;      // Cap on percentage discounts
  final DateTime expiresAt;
  final bool isActive;

  double calculateDiscount(double orderTotal) {
    if (orderTotal < minOrderValue) return 0;
    if (discountType == DiscountType.fixed) return discountValue;
    final pct = orderTotal * (discountValue / 100);
    return maxDiscount != null ? pct.clamp(0, maxDiscount!) : pct;
  }

  bool get isExpired => DateTime.now().isAfter(expiresAt);
  bool get isValid => isActive && !isExpired;
}
```
